\section{Теоретическая часть}

\subsection{Основные понятия теории вероятностей}

\begin{definition}
Пространством элементарных исходов $\Omega$ называется множество всех возможных исходов некоторого случайного эксперимента. Каждый элементарный исход $\omega \in \Omega$ соответствует одному возможному результату эксперимента.
\end{definition}

\begin{definition}
Случайным событием называется любое подмножество $A \subseteq \Omega$. Говорят, что событие $A$ произошло, если в результате эксперимента наступил элементарный исход $\omega \in A$.
\end{definition}

\begin{definition}
События называются несовместными, если они не могут произойти одновременно, то есть $A \cap B = \varnothing$.
\end{definition}

\begin{definition}
Вероятностью называется числовая функция $P$, определённая на множестве событий $\mathcal{F}$ ($\sigma$-алгебре), удовлетворяющая трём аксиомам Колмогорова:
\begin{enumerate}
    \item Аксиома неотрицательности: $P(A) \ge 0$ для любого события $A$.
    \item Аксиома нормированности: $P(\Omega) = 1$.
    \item Аксиома счётной аддитивности: для любой последовательности попарно несовместных событий $A_1, A_2, \dots$ выполняется:
    \[
    P\left(\bigcup_{i=1}^{\infty} A_i\right) = \sum_{i=1}^{\infty} P(A_i).
    \]
\end{enumerate}
\end{definition}

\begin{definition}
Если пространство элементарных исходов $\Omega$ состоит из $N$ равновозможных исходов, а событию $A$ благоприятствует $M$ исходов, то вероятность события $A$ вычисляется по формуле:
\begin{equation}
    P(A) = \frac{M}{N}. \tag{1}
\end{equation}
\end{definition}

\begin{definition}
Вероятность события $A$ есть предел частоты его появления в серии из $n$ независимых испытаний при $n \to \infty$:
\begin{equation}
    P(A) = \lim_{n \to \infty} \frac{m_n}{n}, \tag{2}
\end{equation}
где $m_n$ — число испытаний, в которых наступило событие $A$.
\end{definition}

\subsection{Теоремы сложения и умножения вероятностей}

\begin{theorem}[Сложение вероятностей для несовместных событий]
Если события $A$ и $B$ несовместны, то вероятность их объединения равна сумме вероятностей:
\begin{equation}
    P(A \cup B) = P(A) + P(B). \tag{3}
\end{equation}
\end{theorem}

\begin{proof}
Поскольку $A$ и $B$ несовместны, $A \cap B = \varnothing$. Тогда число исходов, благоприятствующих $A \cup B$, равно сумме исходов, благоприятствующих $A$ и $B$ по отдельности. При классическом определении вероятности:
\[
P(A \cup B) = \frac{M_A + M_B}{N} = \frac{M_A}{N} + \frac{M_B}{N} = P(A) + P(B).
\]
В общем случае это следует из аксиомы аддитивности для конечного числа несовместных событий.
\end{proof}

\begin{theorem}[Сложение вероятностей для совместных событий]
Для любых двух событий $A$ и $B$ справедлива формула:
\begin{equation}
    P(A \cup B) = P(A) + P(B) - P(A \cap B). \tag{4}
\end{equation}
\end{theorem}

\begin{proof}
Представим $A \cup B$ как объединение трёх несовместных событий: $A \setminus B$, $B \setminus A$ и $A \cap B$. Тогда
\[
P(A \cup B) = P(A \setminus B) + P(B \setminus A) + P(A \cap B).
\]
Выразим $P(A \setminus B) = P(A) - P(A \cap B)$ и $P(B \setminus A) = P(B) - P(A \cap B)$. Подставляя, получаем:
\[
P(A \cup B) = P(A) - P(A \cap B) + P(B) - P(A \cap B) + P(A \cap B) = P(A) + P(B) - P(A \cap B).
\]
\end{proof}

\begin{remark}
Для трёх событий формула имеет вид:
\begin{equation}
    P(A \cup B \cup C) = P(A) + P(B) + P(C) - P(A \cap B) - P(A \cap C) - P(B \cap C) + P(A \cap B \cap C). \tag{5}
\end{equation}
Доказательство проводится аналогично с помощью метода включений-исключений.
\end{remark}

\begin{definition}
Условной вероятностью события $A$ при условии, что событие $B$ произошло ($P(B) > 0$), называется величина:
\begin{equation}
    P(A|B) = \frac{P(A \cap B)}{P(B)}. \tag{6}
\end{equation}
\end{definition}

\begin{theorem}[Умножение вероятностей]
Для любых событий $A$ и $B$:
\begin{equation}
    P(A \cap B) = P(A) \cdot P(B|A) = P(B) \cdot P(A|B). \tag{7}
\end{equation}
\end{theorem}

\begin{proof}
Следует непосредственно из определения условной вероятности: домножая обе части равенства $P(A|B) = \frac{P(A \cap B)}{P(B)}$ на $P(B)$, получаем $P(A \cap B) = P(B) \cdot P(A|B)$. Аналогично для $P(B|A)$.
\end{proof}

Для произвольного числа событий формула обобщается:
\begin{equation}
    P(A_1 \cap A_2 \cap \dots \cap A_n) = P(A_1) \cdot P(A_2|A_1) \cdot P(A_3|A_1 \cap A_2) \cdot \dots \cdot P(A_n|A_1 \cap \dots \cap A_{n-1}). \tag{8}
\end{equation}

\begin{definition}
События $A$ и $B$ называются независимыми, если:
\begin{equation}
    P(A \cap B) = P(A) \cdot P(B). \tag{9}
\end{equation}
События $A_1, A_2, \dots, A_n$ называются независимыми в совокупности, если для любого набора индексов $i_1, \dots, i_k$ выполняется:
\begin{equation}
    P(A_{i_1} \cap \dots \cap A_{i_k}) = P(A_{i_1}) \cdot \dots \cdot P(A_{i_k}). \tag{10}
\end{equation}
\end{definition}

\begin{theorem}[Формула полной вероятности]
Пусть события $H_1, H_2, \dots, H_n$ образуют полную группу попарно несовместных событий (гипотезы). Тогда для любого события $A$:
\begin{equation}
    P(A) = \sum_{i=1}^{n} P(H_i) \cdot P(A|H_i). \tag{11}
\end{equation}
\end{theorem}

\begin{proof}
Так как гипотезы образуют полную группу, $\bigcup_{i=1}^n H_i = \Omega$, и они попарно несовместны. Событие $A$ можно представить как объединение несовместных событий $A \cap H_i$, $i=1,\dots,n$. По аксиоме аддитивности:
\[
P(A) = \sum_{i=1}^n P(A \cap H_i).
\]
Применяя теорему умножения к каждому слагаемому, получаем $P(A \cap H_i) = P(H_i) \cdot P(A|H_i)$. Следовательно,
\[
P(A) = \sum_{i=1}^n P(H_i) \cdot P(A|H_i).
\]
\end{proof}

\begin{theorem}[Формула Байеса]
При тех же условиях:
\begin{equation}
    P(H_k|A) = \frac{P(H_k) \cdot P(A|H_k)}{\sum_{i=1}^{n} P(H_i) \cdot P(A|H_i)}. \tag{12}
\end{equation}
\end{theorem}

\begin{proof}
По определению условной вероятности:
\[
P(H_k|A) = \frac{P(H_k \cap A)}{P(A)}.
\]
Используя теорему умножения для числителя и формулу полной вероятности для знаменателя, получаем требуемое.
\end{proof}

\subsection{Дискретные случайные величины}

\begin{definition}
Случайной величиной называется функция $\xi : \Omega \to \mathbb{R}$, определённая на пространстве элементарных исходов $\Omega$.
\end{definition}

\begin{definition}
Случайная величина называется дискретной, если множество её значений конечно или счётно.
\end{definition}

\begin{definition}
Законом распределения дискретной случайной величины $\xi$ называется соответствие между возможными значениями $x_i$ и их вероятностями $p_i = P(\xi = x_i)$. Обычно задаётся в виде таблицы:
\[
\begin{array}{c|cccc}
x_i & x_1 & x_2 & \dots & x_n \\
\hline
p_i & p_1 & p_2 & \dots & p_n
\end{array}
\]
где $\sum_{i=1}^{n} p_i = 1$ и $p_i \ge 0$.
\end{definition}

\begin{definition}
Математическим ожиданием дискретной случайной величины $\xi$ называется число:
\begin{equation}
    E[\xi] = \sum_{i=1}^{n} x_i p_i. \tag{13}
\end{equation}
\end{definition}

Свойства математического ожидания:
\begin{enumerate}
    \item $E[C] = C$ для константы $C$;
    \item $E[C \cdot \xi] = C \cdot E[\xi]$;
    \item $E[\xi + \eta] = E[\xi] + E[\eta]$;
    \item Если $\xi$ и $\eta$ независимы, то $E[\xi \cdot \eta] = E[\xi] \cdot E[\eta]$.
\end{enumerate}

\begin{definition}
Дисперсией случайной величины $\xi$ называется число:
\begin{equation}
    D[\xi] = E[(\xi - E[\xi])^2] = \sum_{i=1}^{n} (x_i - E[\xi])^2 p_i. \tag{14}
\end{equation}
Удобная формула для вычисления:
\begin{equation}
    D[\xi] = E[\xi^2] - (E[\xi])^2. \tag{15}
\end{equation}
\end{definition}

Свойства дисперсии:
\begin{enumerate}
    \item $D[C] = 0$ для константы $C$;
    \item $D[C \cdot \xi] = C^2 \cdot D[\xi]$;
    \item $D[\xi + C] = D[\xi]$;
    \item Для независимых $\xi$ и $\eta$: $D[\xi + \eta] = D[\xi] + D[\eta]$.
\end{enumerate}

\begin{definition}
Средним квадратическим отклонением называется
\begin{equation}
    \sigma[\xi] = \sqrt{D[\xi]}. \tag{16}
\end{equation}
\end{definition}

\begin{definition}
Коэффициентом асимметрии называется величина, характеризующая асимметрию распределения:
\begin{equation}
    \gamma_1 = \frac{E[(\xi - E\xi)^3]}{\sigma^3}. \tag{17}
\end{equation}
При $\gamma_1 > 0$ правый хвост распределения тяжелее левого, при $\gamma_1 < 0$ — наоборот. Для симметричных распределений $\gamma_1 = 0$.
\end{definition}

\begin{definition}
Коэффициентом эксцесса называется величина:
\begin{equation}
    \gamma_2 = \frac{E[(\xi - E\xi)^4]}{\sigma^4} - 3. \tag{18}
\end{equation}
Для нормального распределения $\gamma_2 = 0$. При $\gamma_2 > 0$ распределение имеет более острую вершину, при $\gamma_2 < 0$ — более плоскую.
\end{definition}

\subsection{Функция распределения}

\begin{definition}
Функцией распределения случайной величины $\xi$ называется функция $F(x) = P(\xi < x)$.
\end{definition}

Свойства функции распределения:
\begin{enumerate}
    \item $F(x)$ — неубывающая функция;
    \item $\lim_{x \to -\infty} F(x) = 0$, $\lim_{x \to +\infty} F(x) = 1$;
    \item $F(x)$ непрерывна слева;
    \item $P(a \le \xi < b) = F(b) - F(a)$.
\end{enumerate}
Для дискретной случайной величины:
\begin{equation}
    F(x) = \sum_{x_i < x} p_i. \tag{19}
\end{equation}

\subsection{Основные дискретные распределения}

\subsubsection{Биномиальное распределение}

\begin{definition}
Проводится $n$ независимых испытаний, в каждом из которых событие $A$ может наступить с вероятностью $p$ (успех) или не наступить с вероятностью $q = 1 - p$ (неудача). Число успехов $X$ имеет биномиальное распределение:
\begin{equation}
    P(X = k) = C_n^k p^k q^{n-k}, \quad k = 0, 1, \dots, n, \tag{20}
\end{equation}
где $C_n^k = \frac{n!}{k!(n-k)!}$ — биномиальный коэффициент.
\end{definition}

Числовые характеристики:
\begin{equation}
    E[X] = np, \quad D[X] = npq. \tag{21}
\end{equation}

\subsubsection{Геометрическое распределение}

\begin{definition}
Случайная величина $X$ имеет геометрическое распределение с параметром $p \in (0, 1]$, если:
\begin{equation}
    P(X = k) = (1 - p)^{k-1} p, \quad k = 1, 2, \dots \tag{22}
\end{equation}
Интерпретация: номер первого успеха в последовательности испытаний Бернулли.
\end{definition}

Числовые характеристики:
\begin{equation}
    E[X] = \frac{1}{p}, \quad D[X] = \frac{1 - p}{p^2}. \tag{23}
\end{equation}

Для конечного геометрического распределения (значения $1, 2, \dots, m$):
\begin{equation}
    P(X = k) = \frac{(1 - p)^{k-1} p}{1 - (1 - p)^m}, \quad k = 1, \dots, m. \tag{24}
\end{equation}

\subsubsection{Равномерное дискретное распределение}

\begin{definition}
Дискретная случайная величина $\xi$ имеет равномерное распределение на множестве $\{x_1, x_2, \dots, x_n\}$, если:
\begin{equation}
    P(\xi = x_i) = \frac{1}{n}, \quad i = 1, 2, \dots, n. \tag{25}
\end{equation}
Для случая $x_i = i$ ($i = 1, \dots, n$):
\begin{equation}
    E\xi = \frac{n + 1}{2}, \quad D\xi = \frac{n^2 - 1}{12}. \tag{26}
\end{equation}
\end{definition}

\subsubsection{Треугольное дискретное распределение}

\begin{definition}
Дискретное треугольное распределение задаётся на множестве целых точек $a, a+1, \dots, b$ с модой в $c$ ($a \le c \le b$):
\begin{equation}
    P(X = x) = \frac{2(x - a + 1)}{(b - a + 1)(c - a + 1)} \quad \text{для } a \le x \le c, \tag{27}
\end{equation}
\begin{equation}
    P(X = x) = \frac{2(b - x + 1)}{(b - a + 1)(b - c + 1)} \quad \text{для } c \le x \le b. \tag{28}
\end{equation}
\end{definition}

\subsection{Комбинаторика}

\begin{definition}
Число перестановок из $n$ различных элементов:
\begin{equation}
    P_n = n! = 1 \cdot 2 \cdot \dots \cdot n. \tag{29}
\end{equation}
\end{definition}

\begin{definition}
Число размещений из $n$ элементов по $k$:
\begin{equation}
    A_n^k = n \cdot (n - 1) \cdot \dots \cdot (n - k + 1) = \frac{n!}{(n - k)!}. \tag{30}
\end{equation}
\end{definition}

\begin{definition}
Число сочетаний из $n$ элементов по $k$:
\begin{equation}
    C_n^k = \binom{n}{k} = \frac{n!}{k!(n - k)!}. \tag{31}
\end{equation}
\end{definition}

\subsection{Элементы теории графов}

\begin{definition}
Графом $G = (V, E)$ называется пара множеств: $V$ — вершины, $E \subseteq V \times V$ — рёбра.
\end{definition}

\begin{definition}
Степенью вершины $v$ называется количество инцидентных ей рёбер: $\deg(v) = |\{u \in V : (v, u) \in E\}|$.
\end{definition}

\begin{definition}
Путём в графе называется последовательность вершин $v_1, v_2, \dots, v_k$, в которой каждые две соседние вершины соединены ребром.
\end{definition}

\begin{definition}
Циклом называется замкнутый путь $v_1, v_2, \dots, v_k, v_1$ без повторяющихся вершин, кроме первой и последней.
\end{definition}

\begin{definition}
Компонентой связности называется максимальное по включению множество вершин, в котором любые две вершины соединены путём.
\end{definition}

\begin{definition}
Деревом называется связный граф без циклов. Для дерева с $n$ вершинами число рёбер равно $n - 1$.
\end{definition}

\begin{definition}
Остовным деревом связного графа называется его подграф, являющийся деревом и содержащий все вершины графа.
\end{definition}

\begin{definition}
Полным графом $K_m$ называется граф на $m$ вершинах, в котором каждая пара различных вершин соединена ребром. Число рёбер в полном графе равно $\frac{m(m - 1)}{2}$.
\end{definition}

\subsection{Парадокс дней рождения}

\begin{theorem}
Для $N$ равновероятных значений вероятность того, что среди $k$ случайно выбранных элементов хотя бы два совпадут, составляет:
\begin{equation}
    P \approx 1 - e^{-\frac{k(k - 1)}{2N}}. \tag{32}
\end{equation}
При $k \approx \sqrt{2N \ln 2}$ вероятность превышает $1/2$. Для $N = 365$ (дни года) $k \approx 23$, что и даёт название парадоксу.
\end{theorem}

\begin{proof}
Точная вероятность отсутствия совпадений (все $k$ элементов различны):
\[
P_{\text{нет}} = \frac{N}{N} \cdot \frac{N-1}{N} \cdot \dots \cdot \frac{N-k+1}{N} = \prod_{i=1}^{k-1} \left(1 - \frac{i}{N}\right).
\]
Используя неравенство $1 - x \le e^{-x}$, получаем:
\[
P_{\text{нет}} \le \exp\left(-\sum_{i=1}^{k-1} \frac{i}{N}\right) = \exp\left(-\frac{k(k-1)}{2N}\right).
\]
Следовательно, вероятность хотя бы одного совпадения:
\[
P = 1 - P_{\text{нет}} \ge 1 - e^{-\frac{k(k-1)}{2N}}.
\]
Для решения уравнения $P = 1/2$ приравниваем $1 - e^{-\frac{k(k-1)}{2N}} = \frac12$, откуда $e^{-\frac{k(k-1)}{2N}} = \frac12$. Логарифмируя, находим $k(k-1) \approx 2N \ln 2$, и при больших $k$ имеем $k \approx \sqrt{2N \ln 2}$.
\end{proof}

\subsubsection{Задание №1. Проверка теоремы сложения вероятностей}

\textbf{Техническое задание:}

Пусть дано множество, содержащее $N$ различных элементов, которые пронумерованы натуральными числами от $1$ до $N$. Будем рассматривать перестановки элементов этого множества. Далее, считаем, что все перестановки равновероятны. Пусть событие $A_i$ — событие, состоящее в том, что для заданной перестановки элементов множества элемент с номером $i$ оказался на $i$-й позиции. Реализуйте приложение, в рамках которого непосредственным подсчётом и аналитически проверьте теорему сложения вероятностей:
\[
P(A_i \cup A_j) = P(A_i) + P(A_j) - P(A_i \cap A_j).
\]
В вашем приложении предоставьте возможность вывода всех благоприятствующих исходов событиям $A_i$, $A_j$. Параметром приложения (передаваемым через командную строку) является количество элементов множества $N$.

\textbf{Математическая модель:}

Общее число равновозможных перестановок равно $N!$. Событие $A_i$ означает, что элемент с номером $i$ зафиксирован на позиции $i$, а остальные $N-1$ элементов могут быть переставлены произвольно. Следовательно, число благоприятствующих исходов для $A_i$ равно $(N-1)!$. Тогда вероятность:
\begin{equation}
    P(A_i) = \frac{(N-1)!}{N!} = \frac{1}{N}. \tag{1.1}
\end{equation}
Аналогично для события $A_j$:
\begin{equation}
    P(A_j) = \frac{1}{N}. \tag{1.2}
\end{equation}

События $A_i$ и $A_j$ совместны. Их пересечение $A_i \cap A_j$ означает, что одновременно элемент $i$ стоит на позиции $i$, а элемент $j$ — на позиции $j$. Остальные $N-2$ элементов переставляются произвольно, поэтому число благоприятствующих исходов равно $(N-2)!$. Вероятность пересечения:
\begin{equation}
    P(A_i \cap A_j) = \frac{(N-2)!}{N!} = \frac{1}{N(N-1)}. \tag{1.3}
\end{equation}

Подставляя (1.1), (1.2) и (1.3) в теорему сложения, получаем аналитическое выражение:
\begin{equation}
    P(A_i \cup A_j) = \frac{1}{N} + \frac{1}{N} - \frac{1}{N(N-1)} = \frac{2}{N} - \frac{1}{N(N-1)}. \tag{1.4}
\end{equation}

\textbf{Алгоритм реализации:}

\begin{enumerate}
    \item Считывается параметр $N$ из командной строки.
    \item Создаётся вектор из чисел $1,2,\dots,N$ и инициализируется счётчики событий $A_i$, $A_j$, $A_i \cap A_j$.
    \item С помощью цикла \texttt{do-while} перебираются все перестановки. Для каждой перестановки проверяются условия: находится ли элемент с номером 1 на первой позиции и элемент с номером 2 на второй позиции (для примера взяты $i=1, j=2$). При выполнении условий увеличиваются соответствующие счётчики.
    \item После завершения перебора вычисляются эмпирические вероятности как отношения счётчиков к $N!$.
    \item Вычисляется теоретическая вероятность объединения по формуле (1.4).
    \item Выводится таблица сравнения эмпирических и аналитических значений.
    \item При указании ключа \texttt{--list} выводятся все перестановки, благоприятствующие каждому событию.
\end{enumerate}

    return perm[i-1] == i;

do {

    if (Ai) cntAi++;           
    if (Aj) cntAj++;            
    if (Ai && Aj) cntInter++;

double pAi   = (double)cntAi / total;

double pInter = (double)cntInter / total;   
double pUnion = pAi + pAj - pInter;

double pUnion_th = 2.0/N - 1.0/(N*(N-1));   
double diff = fabs(pUnion - pUnion_th);

\subsubsection{Задание №2. Моделирование эпидемии}

\textbf{Техническое задание:}

Некоторое заболевание от человека к человеку передается контактным путем. Вероятность заразиться им при контакте равна $p_1 \in (0; 1]$. Заразившийся может излечиться с вероятностью $p_2 \in [0; 1]$. Вспышкой болезни будем называть случайное возникновение заболевания у какого-либо человека. Смоделируйте распространение заболевания, если у вас имеются данные о людях, о их знакомстве друг с другом. Формат входных данных определите самостоятельно. Для демонстрации работы вашей программы тестовые данные можно взять по ссылке: \url{https://habr.com/ru/post/148162/}. По полученным результатам моделирования реализуйте возможности поиска: всех не заразившихся людей; всех исцелившихся людей; поиска исцелившихся людей, окружение которых не исцелилось; определения не заразившихся людей, если всё их окружение заразилось. Считаем окружением человека всех тех, с кем он непосредственно знаком. Организуйте удобный пользовательский интерфейс (Web/Desktop), предоставляющий возможности: загрузки файла с входными данными, конфигурированием значений $p_1$ и $p_2$, запуском/остановкой моделирования, наглядного вывода результатов поиска по результатам моделирования.

\textbf{Математическая модель:}

Модель представляет собой дискретный по времени марковский процесс на графе знакомств. Каждый человек (вершина графа) в каждый момент времени находится в одном из трёх состояний:
\begin{itemize}
    \item $S$ (Susceptible) – здоровый, восприимчивый к инфекции;
    \item $I$ (Infected) – заражённый и заразный;
    \item $R$ (Recovered) – исцелившийся (иммунитет).
\end{itemize}

На каждом шаге моделирования (дискретное время) выполняются следующие переходы:
\begin{enumerate}
    \item Каждый заражённый ($I$) переходит в состояние $R$ с вероятностью $p_2$:
    \begin{equation}
        P(I \to R) = p_2. \tag{2.1}
    \end{equation}
    \item Каждый здоровый ($S$), имеющий $k$ заражённых соседей, заражается с вероятностью:
    \begin{equation}
        P(S \to I \mid k \text{ заражённых соседей}) = 1 - (1 - p_1)^k. \tag{2.2}
    \end{equation}
    Это соответствует тому, что заражение происходит, если хотя бы один контакт с заражённым оказался успешным (независимые испытания).
\end{enumerate}

Моделирование продолжается до тех пор, пока не исчезнут все заражённые ($I = \varnothing$).

По окончании моделирования реализованы запросы:
\begin{itemize}
    \item Все люди, оставшиеся в состоянии $S$ (не заразившиеся).
    \item Все люди, перешедшие в состояние $R$ (исцелившиеся).
    \item Исцелившиеся, у которых хотя бы один сосед не исцелился (т.е. сосед в состоянии $S$ или $I$).
    \item Не заразившиеся, у которых все соседи были заражены (состояние $I$) в некоторый момент.
\end{itemize}

\textbf{Алгоритм реализации:}

\begin{enumerate}
    \item Загрузка графа из текстового файла: считывается количество вершин, имена вершин, затем список рёбер.
    \item Инициализация начального состояния: выбирается случайная вершина, которая становится заражённой.
    \item Основной цикл моделирования:
        \begin{itemize}
            \item Для каждого заражённого перебираются его соседи; если сосед здоров, то с вероятностью $p_1$ он заражается.
            \item После распространения инфекции каждый заражённый с вероятностью $p_2$ выздоравливает и переходит в состояние $R$.
            \item Цикл продолжается, пока есть хотя бы один заражённый.
        \end{itemize}
    \item После завершения моделирования выполняются поисковые запросы по накопленным данным.
    \item Графический интерфейс отображает состояние каждого человека цветом, позволяет задавать параметры, загружать граф и запускать моделирование.
\end{enumerate}

bool step() {
    if (infected.empty()) return false;
    vector<int> current = infected;
    for (int v : current) {
        for (int nb : adj[v]) {
            if (people[nb].state == HEALTHY && rand() < p1) {
                people[nb].state = INFECTED;
                infected.push_back(nb);
            }
        }
    }
    for (int v : current) {
        if (people[v].state == INFECTED && rand() < p2) {
            people[v].state = RECOVERED;
            infected.erase(remove(infected.begin(), infected.end(), v), infected.end());
        }
    }
    return !infected.empty();
}

vector<int> notInfected() const {
    vector<int> res;
    for (int i = 0; i < people.size(); ++i)
        if (people[i].state == HEALTHY) res.push_back(i);
    return res;
}

vector<int> recoveredWithNonRecoveredNeighbors() const {
    vector<int> res;
    for (int v : recovered()) {
        bool hasNonRec = false;
        for (int nb : adj[v])
            if (people[nb].state != RECOVERED) { hasNonRec = true; break; }
        if (hasNonRec) res.push_back(v);
    }
    return res;
}

\begin{equation}
    P(L \to L+k) = 52 \cdot 10^{-2-k}, \quad k \in \mathbb{N}. \tag{3.2}
\end{equation}

При нескольких модулях качества вероятности повышения суммируются аддитивно. Оставшаяся вероятность (до 1) соответствует сохранению текущего уровня качества.

Поступление компонентов уровня 1 происходит непрерывно со скоростями $s_i$ (единиц в единицу времени). Компоненты более высоких уровней (2–5) могут быть получены только как результат сборки (продукт) или переработки (компоненты).

Процесс переработки возвращает 25\% исходных компонентов: для каждого компонента рецепта количество возвращаемых единиц равно $\lfloor \text{recipe}_i / 4 \rfloor$.

Процесс моделирования завершается, когда накоплено 25 продуктов уровня 5.

\textbf{Алгоритм реализации:}

\begin{enumerate}
    \item Читаются параметры из командной строки: $AM, RM, n$, рецепт (список количеств компонентов), $t_{\text{сб}}, t_{\text{пр}}, s_1, \dots, s_n$.
    \item Создаётся пул сборочных автоматов и автоматов переработки (по 4 модуля качества в каждом).
    \item В цикле дискретного времени:
        \begin{itemize}
            \item Пополняются запасы компонентов уровня 1 согласно скоростям $s_i$ (с накоплением дробной части).
            \item Запускаются свободные автоматы на сборку при наличии достаточного количества компонентов.
            \item Запускаются свободные автоматы на переработку при наличии готовых продуктов (уровня 3 и выше).

\end{enumerate}

        if (base[i] + (int)high[i].size() < recipe[i]) return false;

        }
        while (need > 0 && base[i] > 0) {
            comps.push_back({-1, 1, time});
            --base[i]; --need;
        }
    }
    return comps;
};

            for (int k = 1; k <= m.mods; ++k) {

                if (bernoulli_distribution(p)(rng)) { L += k; break; }
            }
        }
        if (L > 5) L = 5;
        record({m.outId, L, m.start});
        m.input.clear();

}

    m.busy = true;

    m.input = {{-1, p.quality, p.time}};
}

    m.remain -= 1.0;
    if (m.remain <= 0.001) {
        m.busy = false;
        int q = m.input[0].quality;
        if (m.mods > 0) {
            for (int k = 1; k <= m.mods; ++k) {

            }

        }
        m.input.clear();
    }

\subsubsection{Задание №4. Двумерное случайное блуждание}

\textbf{Техническое задание:}

Пусть задана координатная плоскость $xOy$. Материальная точка стартует с оси ординат из точки $(0; h)$ с шагом $1$ по оси $Y$. Закон движения точки описывается уравнениями: по оси абсцисс $X_{t+1} = X_t + s$, где $s$ — случайная величина, принимающая значения $s_1, s_2, \dots, s_n$. Смоделируйте движение материальной точки для различных законов распределения случайной величины $s$ (равномерное, биномиальное, конечное геометрическое, дискретное треугольное). В оконном/web-приложении отобразите графически полученную траекторию; также обеспечьте хранение и просмотр лог-записей для последних $K$ моделирований. Эмпирически подсчитайте вероятность пересечений оси абсцисс, предусмотрите случай, когда $l = 0$. Обеспечьте настройку параметров $h, n, s_1, \dots, s_n, K, l$, а также настройку закона распределения случайной величины $s$ на основе конфигурационного файла в формате JSON.

\textbf{Математическая модель:}

Траектория точки задаётся рекуррентными соотношениями:
\begin{equation}
    X_{t+1} = X_t + s_t, \quad Y_{t+1} = Y_t + 1, \tag{4.1}
\end{equation}
где $s_t$ – значение случайной величины $s$ на шаге $t$, выбираемое из множества значений в соответствии с заданным законом распределения. Начальные условия: $X_0 = 0$, $Y_0 = h$.

Точка пересекает ось абсцисс, если существует такой шаг $t$, что $Y_t \le 0$ (поскольку шаг по $Y$ равен +1, это эквивалентно тому, что $Y$ становится неположительным). Число пересечений определяется как количество смен знака $Y_t$ (или переход через ноль).

Для каждого закона распределения проводится $N_{\text{exp}}$ экспериментов, и эмпирическая вероятность пересечения вычисляется как:
\begin{equation}
    P_{\text{пересеч}} = \frac{\text{число траекторий с пересечением}}{N_{\text{exp}}}. \tag{4.2}
\end{equation}

Параметры модели: $h$ – начальная высота, $n$ – число шагов, $K$ – число сохраняемых логов, $l$ – флаг сохранения логов ($l=0$ – не сохранять).

\textbf{Алгоритм реализации:}

\begin{enumerate}
    \item Читаются параметры из JSON-файла (поля: \texttt{h}, \texttt{n}, \texttt{Y}, \texttt{K}, \texttt{l}, \texttt{distribution}, \texttt{params}).
    \item Для каждого закона распределения (равномерное, биномиальное, геометрическое, треугольное) генерируется траектория длины $n$.
    \item Строится график траектории с помощью \texttt{QChart}.
    \item Сохраняются логи последних $K$ траекторий в таблице.
    \item По кнопке "Estimate Prob" оценивается вероятность заданного числа пересечений методом Монте-Карло (10000 экспериментов).
\end{enumerate}

        case 0: s = unif(rng); break;
        case 1: s = binom(rng); break;
        case 2: s = geom(rng); break;
        case 3: {

    wasPos = nowPos;

\subsubsection{Задание №5. Одномерное случайное блуждание с возвратом}

\textbf{Техническое задание:}

Рассмотрим вещественную ось $Ox$ и начало координат. В начальный момент времени материальная точка стартует из 0. С вероятностью $p$ перемещается на $s_+$ вправо и с вероятностью $q = 1 - p$ перемещается на $s_-$ влево. Пусть $N$ — число шагов, за которые частица вернулась в начало координат. Эмпирически подсчитайте вероятности для различных значений $N \le 232$. Обеспечьте настройку параметров $p, q, N, s_+, s_-$ на основе конфигурационного файла в формате JSON.

\textbf{Математическая модель:}

Положение частицы после $t$ шагов описывается суммой случайных приращений:
\begin{equation}
    X_t = \sum_{i=1}^{t} \Delta_i, \quad \Delta_i = 
    \begin{cases}
        s_+, & \text{с вероятностью } p, \\
        -s_-, & \text{с вероятностью } q = 1-p.
    \end{cases} \tag{5.1}
\end{equation}

Время первого возвращения в нуль определяется как:
\begin{equation}
    N = \min\{ t > 0 : X_t = 0 \}. \tag{5.2}
\end{equation}

Для симметричного случая ($p=q=0.5$, $s_+=s_-=1$) известно аналитическое выражение для вероятности возврата на шаге $2k$:
\begin{equation}
    P(N = 2k) = \frac{1}{2k-1} \binom{2k}{k} \frac{1}{2^{2k}}. \tag{5.3}
\end{equation}

В общем случае (несимметричный, разные длины шагов) распределение находится эмпирически методом Монте-Карло.

\textbf{Алгоритм реализации:}

\begin{enumerate}
    \item Читаются параметры $p, q, s_+, s_-, \text{maxN}, \text{num\_simulations}$ из JSON-файла.
    \item Проводится заданное число экспериментов, в каждом из которых генерируется последовательность шагов до первого возврата в 0 или до достижения максимального числа шагов $\text{maxN}$.
    \item Подсчитывается количество возвратов на каждом шаге.
    \item Выводятся эмпирические вероятности $P(N = n)$ для $n = 1, \dots, \text{maxN}$, а также накопленная вероятность и вероятность невозвращения за $\text{maxN}$ шагов.
\end{enumerate}

}
double never = 1.0 - cum;

\subsubsection{Задание №6. Генерация ступенчатых фигур}

\textbf{Техническое задание:}

Пусть дан отрезок $[0; M]$, $M > 0$. Этот отрезок разделен на отрезки длины $h$. На каждом из маленьких отрезков задана случайная величина $\xi$, которая принимает случайное значение из множества $\{0, \tau, 2\tau, \dots, n\tau\}$. Когда на каждом из отрезков определено конкретное значение случайной величины $\xi$, на исходном отрезке возникает ступенчатая фигура. Для различных законов распределения $\xi$ (равномерное, биномиальное, конечное геометрическое, дискретное треугольное) проведите $N$ генераций ступенчатых фигур. Эмпирически подсчитайте вероятность получить ступенчатую фигуру в виде строго возрастающей лестницы. В рамках оконного или web-приложения обеспечьте возможность отображения графиков всевозможных ступенчатых фигур для заданных параметров, с возможностью навигации по ним посредством механизма пагинации. Обеспечьте настройку параметров на основе конфигурационного файла в формате JSON.

\textbf{Математическая модель:}

Отрезок $[0; M]$ разбивается на $m = M/h$ частей. На каждой части $i$ ($i = 0, \dots, m-1$) задаётся значение $\xi_i$ в соответствии с выбранным законом распределения:
\begin{equation}
    \xi_i \in \{0, \tau, 2\tau, \dots, n\tau\}, \quad i = 0, 1, \dots, m-1. \tag{6.1}
\end{equation}

Ступенчатая фигура представляет собой набор точек $(x_i, y_i)$, где $x_i = i \cdot h$, $y_i = \xi_i$.

Фигура называется строго возрастающей лестницей, если выполняется условие:
\begin{equation}
    \xi_0 < \xi_1 < \dots < \xi_{m-1}. \tag{6.2}
\end{equation}

Для каждого закона распределения проводится $N$ генераций, и эмпирическая вероятность вычисляется как:
\begin{equation}
    P_{\text{возр}} = \frac{\text{число возрастающих фигур}}{N}. \tag{6.3}
\end{equation}

\textbf{Алгоритм реализации:}

\begin{enumerate}
    \item Поддерживаются четыре закона распределения: равномерное, биномиальное, конечное геометрическое, дискретное треугольное.
    \item Позволяется задавать параметры: количество шагов $m$, шаг $\tau$, максимальное значение $n$, число генераций $N$.
    \item Генерируются $N$ ступенчатых фигур, отображается текущая фигура с возможностью пагинации (кнопки "Prev" / "Next" с шагом 10).
    \item Вычисляется и отображается эмпирическая вероятность получения строго возрастающей лестницы.
    \item Поддерживается загрузка конфигурации из JSON-файла.
\end{enumerate}

\begin{lstlisting}[caption={Основной алгоритм генерации ступенчатых фигур}, label=lst6]
int64_t generateXi() {
    if (m_distributionType == "uniform") {
        std::uniform_int_distribution<int64_t> dist(0, m_n);
        return dist(m_rng);
    }
    else if (m_distributionType == "binomial") {
        int trials = m_distributionParams.value("trials").toInt(10);
        double p = m_distributionParams.value("p").toDouble(0.5);
        std::binomial_distribution<int> binom(trials, p);
        return binom(m_rng) % (m_n + 1);
    }
    else if (m_distributionType == "finite_geometric") {
        double p = m_distributionParams.value("p").toDouble(0.5);
        std::vector<double> weights(m_n + 1);
        double sum = 0.0;
        for (int i = 0; i <= m_n; ++i) {
            weights[i] = std::pow(1.0 - p, i) * p;
            sum += weights[i];
        }
        for (auto& w : weights) w /= sum;
        std::discrete_distribution<int> dist(weights.begin(), weights.end());
        return dist(m_rng);
    }
    else if (m_distributionType == "discrete_triangular") {
        int a = m_distributionParams.value("a").toInt(0);
        int b = m_distributionParams.value("b").toInt(m_n);
        int c = m_distributionParams.value("c").toInt((a + b) / 2);
        std::vector<double> weights(m_n + 1, 0.0);
        for (int i = a; i <= b; ++i) {
            if (i <= c) weights[i] = (double)(i - a) / (c - a + 1e-9);
            else weights[i] = (double)(b - i) / (b - c + 1e-9);
        }
        double sum = 0.0;
        for (auto& w : weights) sum += w;
        for (auto& w : weights) w /= sum;
        std::discrete_distribution<int> dist(weights.begin(), weights.end());
        return dist(m_rng);
    }
    return 0;
}

\subsubsection{Задание №7. Парадокс дней рождения и атака на шифр Вернама}

\textbf{Техническое задание:}

Пусть дана случайная величина $\xi$, которая принимает значения из заданного конечного множества мощности $N$ с одинаковыми вероятностями. Пусть, далее, множества $X, Y$, которые состоят из $k < N$ значений случайной величины $\xi$. Пусть, далее, $R(k, N) = P\{X \cap Y \neq \varnothing\}$. Докажите, что $R(k, N) \ge 1 - e^{-\frac{k^2}{N}}$. Найдите решение неравенства $R(k, N) > 1/2$. Пусть дана шифрующая функция $E(m)$ алгоритма Вернама, где $m$ — 16-битный открытый текст. Пусть, далее, $X_1, X_2$ — множества из $k$ 16-битных элементов. При каком $k$ существует $x_1 \in X_1, x_2 \in X_2 : E(x_1) = E(x_2)$? Используйте полученное равенство для атаки на $E(m)$.

\textbf{Математическая модель:}

Пусть $\xi$ — случайная величина, принимающая значения из конечного множества мощности $N$ с одинаковыми вероятностями (равномерное распределение). Множества $X$ и $Y$ состоят из $k < N$ значений случайной величины $\xi$, выбранных случайно и независимо. Определим:
\begin{equation}
    R(k, N) = P\{X \cap Y \neq \varnothing\}.
\end{equation}

Требуется доказать нижнюю оценку:
\begin{equation}
    R(k, N) \ge 1 - e^{-\frac{k^2}{N}},
\end{equation}
найти решение неравенства $R(k, N) > 1/2$ и применить результат к атаке на шифр Вернама. Данная задача является обобщением парадокса дней рождения.

\textbf{Доказательство оценки:}

\textit{Шаг 1. Точная комбинаторная формула.}

Вероятность того, что множества $X$ и $Y$ не пересекаются, вычисляется точно с использованием числа сочетаний. При фиксированном $X$ множество $Y$ должно выбираться из оставшихся $N-k$ элементов. Следовательно:
\begin{equation}
    P(X \cap Y = \varnothing) = \frac{C_{N-k}^k}{C_N^k}.
\end{equation}

\textit{Шаг 2. Преобразование к произведению.}

Раскладывая биномиальные коэффициенты, получаем:
\begin{equation}
    \frac{C_{N-k}^k}{C_N^k} = \prod_{i=0}^{k-1} \frac{N-k-i}{N-i} = \prod_{i=0}^{k-1} \left(1 - \frac{k}{N-i}\right).
\end{equation}

\textit{Шаг 3. Применение неравенства $1-x \le e^{-x}$.}

Для любого $x$ справедливо $1-x \le e^{-x}$. Применяя это неравенство к каждому сомножителю:
\begin{equation}
    \prod_{i=0}^{k-1} \left(1 - \frac{k}{N-i}\right) \le \exp\left(-\sum_{i=0}^{k-1} \frac{k}{N-i}\right).
\end{equation}

\textit{Шаг 4. Оценка суммы.}

Каждый член суммы $\frac{1}{N-i}$ не меньше $\frac{1}{N}$, так как $N-i \le N$. Поэтому:
\begin{equation}
    \sum_{i=0}^{k-1} \frac{1}{N-i} \ge \frac{k}{N}.
\end{equation}

\textit{Шаг 5. Итоговая оценка.}

Подставляя полученную оценку:
\begin{equation}
    P(X \cap Y = \varnothing) \le \exp\left(-k \cdot \frac{k}{N}\right) = e^{-k^2/N}.
\end{equation}

\textit{Шаг 6. Переход к $R(k, N)$.}

Поскольку $R(k, N) = 1 - P(X \cap Y = \varnothing)$, окончательно имеем:
\begin{equation}
    R(k, N) \ge 1 - e^{-k^2/N}.
\end{equation}

Что и требовалось доказать.

\textbf{Решение неравенства:}

Решим неравенство $R(k, N) > 1/2$:
\begin{equation}
    1 - e^{-k^2/N} > \frac{1}{2} \Rightarrow e^{-k^2/N} < \frac{1}{2}.
\end{equation}

Логарифмируя обе части:
\begin{equation}
    -\frac{k^2}{N} < -\ln 2 \Rightarrow k^2 > N \ln 2.
\end{equation}

Отсюда:
\begin{equation}
    k > \sqrt{N \ln 2}.
\end{equation}

Для $N = 2^{16} = 65536$ получаем численную оценку:
\begin{equation}
    k > \sqrt{65536 \cdot 0.693147} \approx \sqrt{45426} \approx 213.1.
\end{equation}

Таким образом, при $k \ge 214$ вероятность коллизии превышает $1/2$, что согласуется с парадоксом дней рождения.

\textbf{Атака на шифр Вернама:}

Шифр Вернама определяется как:
\begin{equation}
    E(m) = m \oplus K,
\end{equation}
где $m$ — 16-битный открытый текст, $K$ — фиксированный ключ той же длины.

Рассмотрим два случайных множества $X_1$ и $X_2$, каждое из $k$ различных 16-битных элементов. Если $k \ge 214$, то согласно доказанной оценке с вероятностью более $1/2$ существует пара $(x_1, x_2) \in X_1 \times X_2$ такая, что:
\begin{equation}
    E(x_1) = E(x_2).
\end{equation}

Подставляя определение шифра:
\begin{equation}
    x_1 \oplus K = x_2 \oplus K \Rightarrow x_1 = x_2.
\end{equation}

Это позволяет злоумышленнику восстановить ключ:
\begin{equation}
    K = x_1 \oplus E(x_1).
\end{equation}

Таким образом, задача поиска коллизий в шифре Вернама сводится к нахождению двух одинаковых элементов в множествах размера $k$, что при $k \ge 214$ становится практической атакой.

\subsubsection{Задание №9. Условная вероятность для монет}

\textbf{Техническое задание:}

В кошельке лежат $n$ монет, каждая из которых имеет две стороны. У $k$ монет орёл на обеих сторонах, у остальных орёл только на одной стороне. Случайным образом выберем монету и четыре раза подряд подбросим её. Какова условная вероятность того, что при четвертом подбрасывании снова выпадет орёл, если при трёх предыдущих попытках тоже выпал орёл? Оцените условную вероятность посредством выполнения экспериментов в рамках моделирующего приложения.

\textbf{Математическая модель:}

Обозначим:
\begin{itemize}
    \item $H$ – событие "выбрана монета с двумя орлами";
    \item $\overline{H}$ – событие "выбрана обычная монета (орёл с одной стороны)";
    \item $O_3$ – событие "при трёх первых бросках выпал орёл";
    \item $O_4$ – событие "при четвёртом броске выпал орёл".
\end{itemize}

Априорные вероятности:
\begin{equation}
    P(H) = \frac{k}{n}, \quad P(\overline{H}) = \frac{n-k}{n}. \tag{9.1}
\end{equation}

Вероятность трёх орлов подряд:
\begin{equation}
    P(O_3) = P(H) \cdot 1 + P(\overline{H}) \cdot \left(\frac{1}{2}\right)^3 = \frac{k}{n} + \frac{n-k}{8n}. \tag{9.2}
\end{equation}

Вероятность четырёх орлов подряд:
\begin{equation}
    P(O_4) = P(H) \cdot 1 + P(\overline{H}) \cdot \left(\frac{1}{2}\right)^4 = \frac{k}{n} + \frac{n-k}{16n}. \tag{9.3}
\end{equation}

Тогда искомая условная вероятность:
\begin{equation}
    P(O_4 \mid O_3) = \frac{P(O_4 \cap O_3)}{P(O_3)} = \frac{P(O_4)}{P(O_3)} = \frac{\frac{k}{n} + \frac{n-k}{16n}}{\frac{k}{n} + \frac{n-k}{8n}} = \frac{8k + n - k}{8k + 2(n-k)}. \tag{9.4}
\end{equation}

Упрощая выражение (9.4):
\begin{equation}
    P(O_4 \mid O_3) = \frac{7k + n}{6k + 2n}. \tag{9.5}
\end{equation}

\textbf{Алгоритм реализации:}

\begin{enumerate}
    \item Принимаются от пользователя количество испытаний $N_{\text{exp}}$, параметры $n$ и $k$.
    \item В каждом эксперименте случайно выбирается монета (двуголовая с вероятностью $k/n$ или обычная с вероятностью $(n-k)/n$).
    \item Подбрасывается выбранная монета 4 раза.
    \item Если первые три броска дали орла, запоминается результат четвёртого броска.
    \item После завершения всех экспериментов вычисляется эмпирическая условная вероятность $P(O_4 \mid O_3)$ и сравнивается с теоретическим значением по формуле (9.5).
\end{enumerate}

\begin{lstlisting}[caption={Основной алгоритм моделирования эксперимента с монетами}, label=lst9]
int totalExperiments;
int n, k;
std::mt19937 rng(std::random_device{}());
std::uniform_real_distribution<double> dist(0.0, 1.0);

int threeHeads = 0;
int fourthHeadGivenThree = 0;

for (int exp = 0; exp < totalExperiments; ++exp) {
    bool isDoubleHeaded = (dist(rng) < (double)k / n);
    
    bool first = (dist(rng) < (isDoubleHeaded ? 1.0 : 0.5));
    bool second = (dist(rng) < (isDoubleHeaded ? 1.0 : 0.5));
    bool third = (dist(rng) < (isDoubleHeaded ? 1.0 : 0.5));
    
    if (first && second && third) {
        threeHeads++;
        bool fourth = (dist(rng) < (isDoubleHeaded ? 1.0 : 0.5));
        if (fourth) fourthHeadGivenThree++;
    }
}

double empiricalProb = (double)fourthHeadGivenThree / threeHeads;
double theoreticalProb = (7.0 * k + n) / (6.0 * k + 2.0 * n);
\end{lstlisting}

\subsubsection{Задание №10. Вероятность взрыва бочки}

\textbf{Техническое задание:}

Вероятность попадания одной пули в бочку с бензином равна $p$. При одном попадании бочка взрывается с вероятностью $p_1$, при двух и более — взрывается наверняка. Найти вероятность того, что бочка взорвётся при $n$ выстрелах. Реализуйте моделирующее приложение для проведения эксперимента.

\textbf{Математическая модель:}

Пусть $X$ — число попаданий в бочку при $n$ выстрелах. Тогда $X$ имеет биномиальное распределение:
\begin{equation}
    P(X = m) = \binom{n}{m} p^m (1-p)^{n-m}, \quad m = 0, 1, \dots, n. \tag{10.1}
\end{equation}

Бочка взрывается в следующих случаях:
\begin{enumerate}
    \item При $m = 0$: взрыва нет.
    \item При $m = 1$: взрыв происходит с вероятностью $p_1$.
    \item При $m \ge 2$: взрыв происходит наверняка (вероятность 1).
\end{enumerate}

Таким образом, вероятность взрыва вычисляется по формуле полной вероятности:
\begin{equation}
    P_{\text{взрыв}} = P(X = 1) \cdot p_1 + P(X \ge 2) \cdot 1. \tag{10.2}
\end{equation}

Поскольку $P(X \ge 2) = 1 - P(X = 0) - P(X = 1)$, получаем:
\begin{equation}
    P_{\text{взрыв}} = \binom{n}{1} p (1-p)^{n-1} \cdot p_1 + \left[1 - (1-p)^n - n p (1-p)^{n-1}\right]. \tag{10.3}
\end{equation}

Упрощая:
\begin{equation}
    P_{\text{взрыв}} = 1 - (1-p)^n - n p (1-p)^{n-1} (1 - p_1). \tag{10.4}
\end{equation}

\textbf{Алгоритм реализации:}

\begin{enumerate}
    \item Задаются входные параметры: вероятность попадания $p$, вероятность взрыва при одном попадании $p_1$, число выстрелов $n$, количество экспериментов $N_{\text{exp}}$.
    \item Для каждого эксперимента выполняется цикл из $n$ итераций, на каждой из которых генерируется случайное число, равномерно распределённое на $[0,1]$. Если оно меньше $p$, то считается попаданием; если число попаданий достигает 2, то эксперимент немедленно завершается с результатом "взрыв".
    \item Если после $n$ выстрелов было ровно одно попадание, то с вероятностью $p_1$ бочка взрывается (генерируется ещё одно случайное число).
    \item Подсчитывается число экспериментов, в которых произошёл взрыв, и вычисляется эмпирическая вероятность как отношение этого числа к общему количеству экспериментов.
    \item Вычисляется теоретическая вероятность по формуле (10.4) и сравнивается с эмпирической.
\end{enumerate}

\begin{lstlisting}[caption={Основной алгоритм моделирования взрыва бочки}, label=lst10]
bool simulateExperiment() {
    int hits = 0;
    for (int i = 0; i < n; ++i) {
        if (dist(rng) < p) {
            hits++;
            if (hits >= 2) return true;
        }
    }
    if (hits == 1) {
        return dist(rng) < p1;
    }
    return false;
}

double theoreticalProbability() const {
    double p0 = std::pow(1.0 - p, n);
    double p1_prob = n * p * std::pow(1.0 - p, n - 1);
    double p2plus = 1.0 - p0 - p1_prob;
    return p1_prob * p1 + p2plus;
}

void runSimulation(int numExperiments) {
    int explosions = 0;
    for (int i = 0; i < numExperiments; ++i) {
        if (simulateExperiment()) explosions++;
    }
    double empirical = (double)explosions / numExperiments;
    double theoretical = theoreticalProbability();
    std::cout << "Theoretical probability: " << theoretical << std::endl;
    std::cout << "Empirical probability: " << empirical << std::endl;
    std::cout << "Difference: " << std::abs(theoretical - empirical) << std::endl;
}
\end{lstlisting}

\subsubsection{Задание №12. Моделирование случайного блуждания}

\textbf{Техническое задание:}

Реализовать оконное или WEB-приложение, моделирующее процесс случайного блуждания точки на $\mathbb{R}$. Для точки должна иметься возможность задать: начальное положение; количество перемещений (шагов) $n$; закон перемещения, заданный в виде объекта дискретной случайной величины (значения ДСВ являются скоростями перемещения точки по направлению возрастания значения координаты точки из $\mathbb{R}$). Также необходимо обеспечить удобство выполнение запуска и остановки процесса моделирования. Во время моделирования пользователь должен иметь возможность наблюдать перемещение точки, причём процесс перемещения точки должен быть непрерывным, скорость перемещения точки должна быть прямо пропорциональна модулю значения шага; каждый шаг точки должен выполняться в течение одной секунды. В результате моделирования также необходимо получить и отобразить дискретную случайную величину, отражающую вероятности попадания точки из начального положения во все конечные за $n$ шагов. Предусмотреть функционал задания закона перемещения точки при помощи десериализации объекта дискретной случайной величины из файлового потока; путь к файлу при этом необходимо получить посредством использования стандартного диалога открытия файла.

\textbf{Математическая модель:}

Положение точки в момент времени $t$ определяется как сумма начальной позиции и накопленных приращений:
\begin{equation}
    x_t = x_0 + \sum_{i=1}^{t} v_i, \tag{12.1}
\end{equation}
где $v_i$ — значения дискретной случайной величины, задающей закон перемещения.

Дискретная случайная величина задаётся таблицей значений и вероятностей:
\[
\begin{array}{c|cccc}
v & v_1 & v_2 & \dots & v_m \\
\hline
p & p_1 & p_2 & \dots & p_m
\end{array}
\]
с условиями $\sum p_i = 1$, $p_i \ge 0$.

Каждый шаг выполняется за 1 секунду, при этом скорость перемещения пропорциональна модулю значения шага.

В результате моделирования строится дискретная случайная величина конечных положений $x_n$ с вероятностями, вычисляемыми эмпирически.

\textbf{Алгоритм реализации:}

\begin{enumerate}
    \item Позволяется задать начальную позицию и количество шагов.
    \item Поддерживается загрузка закона перемещения из JSON-файла (формат: значения и соответствующие вероятности).
    \item При запуске моделирования выполняются шаги с интервалом 1 секунда, анимируя движение точки на графике.
    \item Отображается текущий шаг, позиция и статус выполнения.
    \item По окончании моделирования показывается таблица с распределением конечных позиций.
\end{enumerate}

void nextStep() {
    if (currentStep >= totalSteps) {
        stop();
        return;
    }
    double step = values[dist(rng)];
    currentPos += step;
    positions.push_back(currentPos);
    currentStep++;
    update();
}

void loadDistribution() {
    QString file = QFileDialog::getOpenFileName(this, "select JSON", "", "JSON (*.json)");
    if (file.isEmpty()) return;
    QFile f(file);
    if (!f.open(QIODevice::ReadOnly)) return;
    QJsonDocument doc = QJsonDocument::fromJson(f.readAll());
    QJsonObject obj = doc.object();
    if (obj.contains("values") && obj.contains("probabilities")) {
        QJsonArray vals = obj["values"].toArray();
        QJsonArray probs = obj["probabilities"].toArray();
        values.clear();
        std::vector<double> probsVec;
        for (int i = 0; i < vals.size(); ++i) {
            values.push_back(vals[i].toDouble());
            probsVec.push_back(probs[i].toDouble());
        }
        dist = std::discrete_distribution<int>(probsVec.begin(), probsVec.end());
    }
}

void showStatistics() {
    std::map<int, int> freq;
    for (double p : positions) {
        freq[static_cast<int>(std::round(p))]++;
    }
    QTableWidget *table = new QTableWidget((int)freq.size(), 2);
    table->setHorizontalHeaderLabels({"Position", "Probability"});
    int row = 0;
    for (const auto& pair : freq) {
        double prob = static_cast<double>(pair.second) / positions.size();
        table->setItem(row, 0, new QTableWidgetItem(QString::number(pair.first)));
        table->setItem(row, 1, new QTableWidgetItem(QString::number(prob, 'f', 4)));
        row++;
    }
}
